cmake_minimum_required(VERSION 3.21)
project(MetadataProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Multimedia)
add_executable(metadataProbe main.cpp ../../src/AudioMetadataReader.cpp ../../src/AudioMetadataReader.h)
target_link_libraries(metadataProbe PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia)
get_filename_component(FUTARI_SOURCE_DIR "../.." ABSOLUTE)
target_compile_definitions(metadataProbe PRIVATE FUTARI_SOURCE_DIR="${FUTARI_SOURCE_DIR}")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/imports/FutariMusic")
configure_file("../../qml/Theme.qml" "imports/FutariMusic/Theme.qml" COPYONLY)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/imports/FutariMusic/qmldir" "module FutariMusic\nsingleton Theme 1.0 Theme.qml\n")
enable_testing()
add_test(NAME metadataUploadRegression COMMAND metadataProbe)
set_tests_properties(metadataUploadRegression PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
